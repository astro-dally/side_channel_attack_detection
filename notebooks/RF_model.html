print("[*] Stage 1: Preprocessing Telemetry Data...")

def load_and_structure_hpc_data(file_path, label):
    """
    Transforms raw 'perf' traces into a supervised learning feature matrix.
    Labels: 0 = Benign Baseline, 1 = Side-Channel Attack (Interference)
    """
    cols = ['timestamp', 'value', 'unit', 'event', 'count', 'percentage', 'ratio', 'note']
    df = pd.read_csv(file_path, comment='#', names=cols)
    
    # Clean non-numeric values and pivot vertical events into horizontal features
    df['value'] = pd.to_numeric(df['value'], errors='coerce')
    df_pivoted = df.pivot_table(index='timestamp', columns='event', values='value').reset_index()
    
    # Append the ground-truth label
    df_pivoted['label'] = label
    return df_pivoted

# Load datasets
df_benign = load_and_structure_hpc_data('benign_data.csv', 0)
df_attack = load_and_structure_hpc_data('attack_data_final.csv', 1)

# Merge datasets and drop incomplete intervals (NaNs)
df_master = pd.concat([df_benign, df_attack], axis=0).dropna().reset_index(drop=True)

print(f"[+] Integration Complete: {len(df_master):,} labeled samples recovered.")
print(f"[+] Features Extracted: {list(df_master.columns[1:-1])}")

print("\n[*] Stage 3: Training Detection Engine...")

X = df_master.drop(['timestamp', 'label'], axis=1)
y = df_master['label']

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.20, stratify=y, random_state=42)

# Random Forest Ensemble
clf = RandomForestClassifier(n_estimators=100, max_depth=15, n_jobs=-1, random_state=42)
clf.fit(X_train, y_train)
y_pred = clf.predict(X_test)
